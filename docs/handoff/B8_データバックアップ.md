# B8: データバックアップ (D1→SQL全件出力・復元)

**用途**: D1データベースの全テーブルをSQL形式で出力してR2に保存。障害時・誤削除時に復元可能。

**特徴**:
- Cloudflare Workers内で完結（外部ツール不要）
- テーブル定義 + 全レコード を1つのSQLファイルに
- R2 に日時付きファイル名で保存
- 管理画面から手動 & Cron自動 の両方対応

---

## 📁 backup ヘルパー

```typescript
// src/lib/backup.ts

export function buildBackupFilename(): string {
  const now = new Date(Date.now() + 9 * 60 * 60 * 1000)  // JST
  const y = now.getUTCFullYear()
  const m = String(now.getUTCMonth() + 1).padStart(2, '0')
  const d = String(now.getUTCDate()).padStart(2, '0')
  const hh = String(now.getUTCHours()).padStart(2, '0')
  const mm = String(now.getUTCMinutes()).padStart(2, '0')
  return `backup_${y}${m}${d}_${hh}${mm}.sql`
}

export async function generateBackupSql(db: D1Database): Promise<string> {
  const tables = ['users','applications','circulation_steps','attachments',
                  'notification_logs','smtp_settings','lineworks_config',
                  'sessions','drive_uploads', /* ... */]

  let sql = `-- Backup generated at ${new Date().toISOString()}\n\n`
  sql += 'PRAGMA foreign_keys = OFF;\n\n'

  for (const table of tables) {
    // CREATE TABLE
    const schema = await db.prepare(`SELECT sql FROM sqlite_master WHERE type='table' AND name = ?`).bind(table).first() as any
    if (!schema) continue
    sql += `DROP TABLE IF EXISTS ${table};\n${schema.sql};\n\n`

    // INSERT
    const rows = await db.prepare(`SELECT * FROM ${table}`).all()
    for (const row of rows.results as any[]) {
      const cols = Object.keys(row)
      const vals = cols.map(c => {
        const v = row[c]
        if (v === null || v === undefined) return 'NULL'
        if (typeof v === 'number') return String(v)
        return `'${String(v).replace(/'/g, "''")}'`
      })
      sql += `INSERT INTO ${table} (${cols.join(',')}) VALUES (${vals.join(',')});\n`
    }
    sql += '\n'
  }

  sql += 'PRAGMA foreign_keys = ON;\n'
  return sql
}

export async function saveBackup(r2: R2Bucket, filename: string, sqlContent: string): Promise<void> {
  await r2.put(`backups/${filename}`, sqlContent, {
    httpMetadata: { contentType: 'application/sql' }
  })
}

export async function listBackups(r2: R2Bucket): Promise<Array<{name: string, size: number, uploaded: Date}>> {
  const list = await r2.list({ prefix: 'backups/' })
  return list.objects.map(o => ({ name: o.key.replace('backups/', ''), size: o.size, uploaded: o.uploaded }))
    .sort((a, b) => b.uploaded.getTime() - a.uploaded.getTime())
}

export async function getBackup(r2: R2Bucket, filename: string): Promise<string | null> {
  const obj = await r2.get(`backups/${filename}`)
  if (!obj) return null
  return await obj.text()
}

export async function deleteBackup(r2: R2Bucket, filename: string): Promise<void> {
  await r2.delete(`backups/${filename}`)
}

export async function pruneOldBackups(r2: R2Bucket, keepDays: number = 30): Promise<number> {
  const list = await r2.list({ prefix: 'backups/' })
  const cutoff = Date.now() - keepDays * 24 * 60 * 60 * 1000
  let deleted = 0
  for (const o of list.objects) {
    if (o.uploaded.getTime() < cutoff) {
      await r2.delete(o.key)
      deleted++
    }
  }
  return deleted
}

export async function restoreFromSql(db: D1Database, sqlContent: string): Promise<void> {
  // セミコロンで分割して1文ずつ実行
  const stmts = sqlContent.split(/;\s*\n/).filter(s => s.trim() && !s.trim().startsWith('--'))
  for (const stmt of stmts) {
    try {
      await db.prepare(stmt).run()
    } catch (e) {
      console.error('restore stmt failed:', stmt.slice(0, 100), e)
      throw e
    }
  }
}
```

admin.post('/backup/create', async (c) => {
  const filename = buildBackupFilename()
  const sqlContent = await generateBackupSql(c.env.DB)
  await saveBackup(c.env.R2, filename, sqlContent)
  return c.redirect('/admin/backup')
})

admin.get('/backup/download/:name', async (c) => {
  const name = c.req.param('name')
  const content = await getBackup(c.env.R2, name)
  if (!content) return c.notFound()
  return new Response(content, {
    headers: {
      'Content-Type': 'application/sql',
      'Content-Disposition': `attachment; filename="${name}"`
    }
  })
})

## ⏰ 定期自動バックアップ (Cron)

```typescript
// src/index.tsx scheduled ハンドラ内
async function scheduled(event, env, ctx) {
  ctx.waitUntil((async () => {
    // 毎日自動バックアップ
    const filename = buildBackupFilename()
    const sqlContent = await generateBackupSql(env.DB)
    await saveBackup(env.R2, filename, sqlContent)

    // 30日以上経過したものは削除
    await pruneOldBackups(env.R2, 30)
  })())
}
```

## 💡 復元時の注意

- 復元前に必ず現在のDBもバックアップする（誤操作の巻き戻し用）
- ID採番のリセットに注意（`AUTO_INCREMENT` の状態も引き継がれる）
- 大量データの場合はタイムアウトに注意（Workers CPU制限50ms〜30秒）
