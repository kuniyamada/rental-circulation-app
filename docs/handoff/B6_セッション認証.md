# B6: セッション認証 (sessions テーブル + Cookie)

**用途**: ログイン状態の維持。Cookieにセッションキーを持たせ、DBにセッション情報を保存

**特徴**:
- Cloudflare Workers 対応（JWT/OAuthより単純）
- HttpOnly Cookie で XSS対策
- パスワードは bcrypt でハッシュ化保存

---

## 📁 テーブル

```sql
CREATE TABLE IF NOT EXISTS sessions (
  id TEXT PRIMARY KEY,                  -- ランダム32文字
  user_id INTEGER NOT NULL,
  expires_at INTEGER NOT NULL,          -- Unix秒
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id)
);
CREATE INDEX idx_sessions_user_id ON sessions(user_id);

CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  employee_number TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,          -- bcrypt hash
  name TEXT NOT NULL,
  email TEXT,
  role TEXT NOT NULL,                   -- 'admin' / 'front' / 'operations' / etc
  is_admin INTEGER DEFAULT 0,
  is_active INTEGER DEFAULT 1
);
```

## 🎯 ログイン処理

```typescript
// src/routes/auth.ts
auth.post('/login', async (c) => {
  const body = await c.req.parseBody() as any
  const { employee_number, password } = body

  const user = await c.env.DB.prepare(
    'SELECT * FROM users WHERE employee_number = ? AND is_active = 1'
  ).bind(employee_number).first() as any

  if (!user || !(await verifyPassword(password, user.password_hash))) {
    return c.html(loginPage('ログイン失敗'))
  }

  // セッション作成
  const sessionId = crypto.randomUUID().replace(/-/g, '')
  const expiresAt = Math.floor(Date.now() / 1000) + 60 * 60 * 24 * 7  // 7日
  await c.env.DB.prepare(
    'INSERT INTO sessions (id, user_id, expires_at) VALUES (?, ?, ?)'
  ).bind(sessionId, user.id, expiresAt).run()

  // Cookie設定
  c.header('Set-Cookie', `session=${sessionId}; Path=/; HttpOnly; SameSite=Strict; Max-Age=${60*60*24*7}${isProduction ? '; Secure' : ''}`)
  return c.redirect('/')
})

auth.post('/logout', async (c) => {
  const sessionId = getSessionIdFromCookie(c.req.header('Cookie'))
  if (sessionId) {
    await c.env.DB.prepare('DELETE FROM sessions WHERE id = ?').bind(sessionId).run()
  }
  c.header('Set-Cookie', 'session=; Path=/; HttpOnly; Max-Age=0')
  return c.redirect('/login')
})
```

## 🔑 セッションヘルパー

```typescript
// src/lib/auth.ts
export function getSessionIdFromCookie(cookieHeader: string | undefined): string | null {
  if (!cookieHeader) return null
  const m = cookieHeader.match(/session=([a-f0-9]+)/)
  return m ? m[1] : null
}

export async function getSessionUser(db: D1Database, sessionId: string | null): Promise<any | null> {
  if (!sessionId) return null
  const row = await db.prepare(`
    SELECT s.user_id as uid, s.expires_at, u.*
    FROM sessions s JOIN users u ON s.user_id = u.id
    WHERE s.id = ? AND s.expires_at > ? AND u.is_active = 1
  `).bind(sessionId, Math.floor(Date.now()/1000)).first() as any
  return row || null
}

// パスワードハッシュ（Web Crypto APIでPBKDF2）
export async function hashPassword(password: string): Promise<string> {
  const salt = crypto.getRandomValues(new Uint8Array(16))
  const key = await crypto.subtle.importKey('raw', new TextEncoder().encode(password), 'PBKDF2', false, ['deriveBits'])
  const hash = await crypto.subtle.deriveBits({ name: 'PBKDF2', salt, iterations: 100000, hash: 'SHA-256' }, key, 256)
  const hashB64 = btoa(String.fromCharCode(...new Uint8Array(hash)))
  const saltB64 = btoa(String.fromCharCode(...salt))
  return `${saltB64}:${hashB64}`
}

export async function verifyPassword(password: string, stored: string): Promise<boolean> {
  const [saltB64, hashB64] = stored.split(':')
  const salt = Uint8Array.from(atob(saltB64), c => c.charCodeAt(0))
  const key = await crypto.subtle.importKey('raw', new TextEncoder().encode(password), 'PBKDF2', false, ['deriveBits'])
  const hash = await crypto.subtle.deriveBits({ name: 'PBKDF2', salt, iterations: 100000, hash: 'SHA-256' }, key, 256)
  const newHashB64 = btoa(String.fromCharCode(...new Uint8Array(hash)))
  return newHashB64 === hashB64
}
```

## 🛡 認証ミドルウェア

```typescript
// 各ルートで使う
const user = await getSessionUser(c.env.DB, getSessionIdFromCookie(c.req.header('Cookie')))
if (!user) return c.redirect('/login')

// admin.ts の共通ミドルウェア
admin.use('*', async (c, next) => {
  const user = await getSessionUser(c.env.DB, getSessionIdFromCookie(c.req.header('Cookie')))
  if (!user) return c.redirect('/login')
  if (!user.is_admin) return c.redirect('/')
  c.set('user' as any, user)
  await next()
})
```
