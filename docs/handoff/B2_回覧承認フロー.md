# B2: 回覧承認フロー (多段承認・Step1→2→3)

**用途**: 送金依頼を複数の承認者に順番に回覧して、全員承認で完了する仕組み

**特徴**:
- 承認ステップ数は柔軟に設定可能（1〜N段）
- 各ステップに指定ユーザーを割り当て
- 順番に通知が回る（Step1完了→Step2へ通知→...）
- 全承認で「完了」、途中で否決/差し戻し/保留可能

## 📁 テーブル設計

### applications テーブル（送金依頼版では requests）
```sql
CREATE TABLE IF NOT EXISTS applications (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  application_number TEXT UNIQUE NOT NULL,   -- 例: REQ-20260918-3408
  title TEXT NOT NULL,                       -- 件名
  applicant_id INTEGER NOT NULL,             -- 申請者(依頼者)
  status TEXT NOT NULL DEFAULT 'circulating', -- circulating/completed/rejected/returned/on_hold
  current_step INTEGER NOT NULL DEFAULT 1,   -- 現在のステップ番号
  -- 送金依頼固有カラム
  amount INTEGER,                            -- 金額
  payee_id INTEGER,                          -- 支払先
  -- 差し戻し情報
  returned_reason TEXT,
  returned_from_step INTEGER,
  returned_by_id INTEGER,
  -- テストモード
  is_test INTEGER DEFAULT 0,
  -- 再申請用
  resubmit_count INTEGER DEFAULT 0,
  original_application_id INTEGER,
  reapply_reason TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (applicant_id) REFERENCES users(id)
);
```

### circulation_steps テーブル（回覧ステップ）
```sql
CREATE TABLE IF NOT EXISTS circulation_steps (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  application_id INTEGER NOT NULL,
  step_number INTEGER NOT NULL,              -- 1, 2, 3, ...
  reviewer_id INTEGER NOT NULL,              -- 承認者
  status TEXT NOT NULL DEFAULT 'pending',    -- pending/approved/rejected/on_hold/returned
  action_comment TEXT,                       -- コメント（否決理由等）
  hold_answer TEXT,                          -- 保留への回答
  acted_at DATETIME,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (application_id) REFERENCES applications(id),
  FOREIGN KEY (reviewer_id) REFERENCES users(id)
);
CREATE INDEX IF NOT EXISTS idx_steps_app_id ON circulation_steps(application_id);
```

## 🎯 フロー全体像

```
[依頼作成]
   ↓
[Step1 承認者A に通知]
   ↓
承認 ─→ [Step2 承認者B に通知]
   ↓                ↓
   否決/差戻し      承認 ─→ [Step3 承認者C に通知]
                                    ↓
                                    承認 ─→ [完了]
                                    否決/差戻し ─→ [依頼者へ通知]
```

## 📝 承認ステップの作成（依頼作成時）

```typescript
async function createCirculationSteps(
  db: D1Database,
  reqId: number,
  applicantId: number,
  reviewer1: number,  // Step1承認者
  reviewer2: number,  // Step2承認者
  reviewer3?: number, // Step3承認者（オプション）
): Promise<void> {
  await db.prepare(
    'INSERT INTO circulation_steps (application_id, step_number, reviewer_id) VALUES (?, 1, ?)'
  ).bind(reqId, reviewer1).run()
  await db.prepare(
    'INSERT INTO circulation_steps (application_id, step_number, reviewer_id) VALUES (?, 2, ?)'
  ).bind(reqId, reviewer2).run()
  if (reviewer3) {
    await db.prepare(
      'INSERT INTO circulation_steps (application_id, step_number, reviewer_id) VALUES (?, 3, ?)'
    ).bind(reqId, reviewer3).run()
  }
}
```

## ✅ 承認処理（POST /:id/review/:stepId）

```typescript
requests.post('/:id/review/:stepId', async (c) => {
  const { id, stepId } = c.req.param()
  const body = await c.req.parseBody() as any
  const { action, comment } = body  // action: approve/reject/hold/return

  const step = await db.prepare(
    'SELECT * FROM circulation_steps WHERE id = ? AND application_id = ?'
  ).bind(stepId, id).first() as any
  if (!step || step.reviewer_id !== user.uid) return c.redirect(`/requests/${id}`)

  const app = await db.prepare(`
    SELECT a.*, u.email as applicant_email, u.name as applicant_name
    FROM applications a JOIN users u ON a.applicant_id = u.id WHERE a.id = ?
  `).bind(id).first() as any

  if (action === 'approve') {
    // ステップを承認済に
    await db.prepare(
      'UPDATE circulation_steps SET status = "approved", acted_at = datetime("now") WHERE id = ?'
    ).bind(stepId).run()

    // 次のステップを探す
    const nextStep = await db.prepare(
      'SELECT * FROM circulation_steps WHERE application_id = ? AND step_number = ? AND status = "pending"'
    ).bind(id, step.step_number + 1).first() as any

    if (nextStep) {
      // 次のステップへ通知
      await db.prepare(
        'UPDATE applications SET current_step = ?, updated_at = datetime("now") WHERE id = ?'
      ).bind(step.step_number + 1, id).run()
      runInBackground(c, async () => {
        await sendNotification(db, 'review_request', nextStep.reviewer_id, {...})
      })
    } else {
      // 全ステップ完了
      await db.prepare(
        'UPDATE applications SET status = "completed", updated_at = datetime("now") WHERE id = ?'
      ).bind(id).run()
      runInBackground(c, async () => {
        await sendNotification(db, 'completed', app.applicant_id, {...})
      })
    }

  } else if (action === 'reject') {
    // 否決処理（詳細はB3）
    ...
  } else if (action === 'return') {
    // 差し戻し処理（詳細はB3）
    ...
  } else if (action === 'hold') {
    // 保留処理（詳細はB3）
    ...
  }

  return c.redirect(`/requests/${id}`)
})
```
